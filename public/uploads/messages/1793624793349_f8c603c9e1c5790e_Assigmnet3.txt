-- Create Database
CREATE DATABASE finance_management_system;

-- Use database
use finance_management_system;

-- 1. Users Table
CREATE TABLE users (
    user_id VARCHAR(50) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL
);

-- 2. Account Types Table
CREATE TABLE account_types (
    account_type_id VARCHAR(50) PRIMARY KEY,
    type_name VARCHAR(100) NOT NULL UNIQUE
);

-- 3. Accounts Table
CREATE TABLE accounts (
    account_id VARCHAR(50) PRIMARY KEY,
    user_id VARCHAR(50) NOT NULL,
    account_type_id VARCHAR(50) NOT NULL,
    balance DECIMAL(12,2) NOT NULL DEFAULT 0.00
);

-- 4. Transaction Types Table
CREATE TABLE transaction_types (
    transaction_type_id VARCHAR(50) PRIMARY KEY,
    type_name VARCHAR(50) NOT NULL UNIQUE
);

-- 5. Categories Table
CREATE TABLE categories (
    category_id VARCHAR(50) PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE
);

-- 6. Subcategories Table
CREATE TABLE subcategories (
    subcategory_id VARCHAR(50) PRIMARY KEY,
    category_id VARCHAR(50) NOT NULL,
    name VARCHAR(100) NOT NULL
);

-- 7. Transactions Table
CREATE TABLE transactions (
    transaction_id VARCHAR(50) PRIMARY KEY,
    account_id VARCHAR(50) NOT NULL,
    category_id VARCHAR(50) NOT NULL,
    transaction_type_id VARCHAR(50) NOT NULL,
    amount DECIMAL(12,2) NOT NULL,
    date TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 8. Budgets Table
CREATE TABLE budgets (
    budget_id VARCHAR(50) PRIMARY KEY,
    user_id VARCHAR(50) NOT NULL,
    category_id VARCHAR(50) NOT NULL,
    limit_amount DECIMAL(12,2) NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE NOT NULL
);

-- 9. Goals Table
CREATE TABLE goals (
    goal_id VARCHAR(50) PRIMARY KEY,
    user_id VARCHAR(50) NOT NULL,
    target_amount DECIMAL(12,2) NOT NULL,
    saved_amount DECIMAL(12,2) NOT NULL DEFAULT 0.00,
    deadline DATE NOT NULL
);


-- 10. Savings Table
CREATE TABLE savings (
    saving_id VARCHAR(50) PRIMARY KEY,
    goal_id VARCHAR(50) NOT NULL,
    amount DECIMAL(12,2) NOT NULL,
    date TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 11. Payment Methods Table
CREATE TABLE payment_methods (
    payment_method_id VARCHAR(50) PRIMARY KEY,
    method_name VARCHAR(100) NOT NULL UNIQUE
);

-- 12. Transaction Payments Table
CREATE TABLE transaction_payments (
    id VARCHAR(50) PRIMARY KEY,
    transaction_id VARCHAR(50) NOT NULL,
    payment_method_id VARCHAR(50) NOT NULL
);

-- Get all accounts of a specific user
SELECT 
    a.account_id,
    u.name AS user_name,
    at.type_name AS account_type,
    a.balance
FROM accounts a
JOIN users u ON a.user_id = u.user_id
JOIN account_types at ON a.account_type_id = at.account_type_id
WHERE u.user_id = 'U001';

-- Get all transactions with category and transaction type
SELECT 
    t.transaction_id,
    a.account_id,
    c.name AS category_name,
    tt.type_name AS transaction_type,
    t.amount,
    t.date
FROM transactions t
JOIN accounts a ON t.account_id = a.account_id
JOIN categories c ON t.category_id = c.category_id
JOIN transaction_types tt ON t.transaction_type_id = tt.transaction_type_id;

-- Get total expenses by category
SELECT 
    c.name AS category_name,
    SUM(t.amount) AS total_amount
FROM transactions t
JOIN categories c ON t.category_id = c.category_id
JOIN transaction_types tt ON t.transaction_type_id = tt.transaction_type_id
WHERE tt.type_name = 'Expense'
GROUP BY c.name;

-- Get user budgets with category names
SELECT 
    b.budget_id,
    u.name AS user_name,
    c.name AS category_name,
    b.limit_amount,
    b.start_date,
    b.end_date
FROM budgets b
JOIN users u ON b.user_id = u.user_id
JOIN categories c ON b.category_id = c.category_id
WHERE u.user_id = 'U001';


-- Get goal progress
SELECT 
    g.goal_id,
    u.name AS user_name,
    g.target_amount,
    g.saved_amount,
    (g.target_amount - g.saved_amount) AS remaining_amount,
    g.deadline
FROM goals g
JOIN users u ON g.user_id = u.user_id;


-- Get savings history for a goal
SELECT 
    s.saving_id,
    g.goal_id,
    s.amount,
    s.date
FROM savings s
JOIN goals g ON s.goal_id = g.goal_id
WHERE g.goal_id = 'G001';

-- Get payment methods used for transactions
SELECT 
    t.transaction_id,
    t.amount,
    pm.method_name
FROM transaction_payments tp
JOIN transactions t ON tp.transaction_id = t.transaction_id
JOIN payment_methods pm ON tp.payment_method_id = pm.payment_method_id;

-- Get transactions between two dates
SELECT *
FROM transactions
WHERE date BETWEEN '2026-01-01' AND '2026-01-31';

-- Get account balance summary for each user
SELECT 
    u.user_id,
    u.name,
    SUM(a.balance) AS total_balance
FROM users u
JOIN accounts a ON u.user_id = a.user_id
GROUP BY u.user_id, u.name;
